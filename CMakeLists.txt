cmake_minimum_required(VERSION 3.20)
project(AetherwildCpp VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aetherwild_options INTERFACE)
target_include_directories(aetherwild_options INTERFACE include)
if(MSVC)
  target_compile_options(aetherwild_options INTERFACE /utf-8 /W4 /permissive-)
else()
  target_compile_options(aetherwild_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

# Feature object targets preserve references inside the final core archive.
set(AETHERWILD_MODULES core runtime physics npc monsters combat inventory
    progression quests gathering crafting economy dungeons social storage network world)
add_library(aetherwild_core STATIC)
target_link_libraries(aetherwild_core PUBLIC aetherwild_options)
foreach(module IN LISTS AETHERWILD_MODULES)
  file(GLOB MODULE_SOURCES CONFIGURE_DEPENDS "src/modules/${module}/*.cpp")
  if(NOT MODULE_SOURCES)
    message(FATAL_ERROR "Empty game module: ${module}")
  endif()
  add_library(aw_${module} OBJECT ${MODULE_SOURCES})
  target_link_libraries(aw_${module} PRIVATE aetherwild_options)
  target_sources(aetherwild_core PRIVATE $<TARGET_OBJECTS:aw_${module}>)
endforeach()
file(GLOB GENERATED_SOURCES CONFIGURE_DEPENDS src/generated/*.cpp)
add_library(aw_generated OBJECT ${GENERATED_SOURCES})
target_link_libraries(aw_generated PRIVATE aetherwild_options)
target_sources(aetherwild_core PRIVATE $<TARGET_OBJECTS:aw_generated>)
add_executable(aetherwild apps/cli.cpp)
target_link_libraries(aetherwild PRIVATE aetherwild_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(aetherwild_tests tests/core_tests.cpp)
  target_link_libraries(aetherwild_tests PRIVATE aetherwild_core)
  add_test(NAME core COMMAND aetherwild_tests)
endif()

add_executable(aetherwild_workstation apps/workstation.cpp)
target_link_libraries(aetherwild_workstation PRIVATE aetherwild_core)
if(WIN32)
  target_link_libraries(aetherwild_workstation PRIVATE ws2_32)
endif()

if(BUILD_TESTING)
 add_executable(aetherwild_runtime_tests tests/runtime_tests.cpp)
 target_link_libraries(aetherwild_runtime_tests PRIVATE aetherwild_core)
 add_test(NAME workstation COMMAND aetherwild_runtime_tests)
endif()


if(BUILD_TESTING)
 add_executable(aetherwild_content_tests tests/content_tests.cpp)
 target_link_libraries(aetherwild_content_tests PRIVATE aetherwild_core)
 add_test(NAME content COMMAND aetherwild_content_tests)
endif()
