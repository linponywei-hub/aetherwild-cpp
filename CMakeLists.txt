cmake_minimum_required(VERSION 3.20)
project(AetherwildCpp VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM CORE_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(aetherwild_core ${CORE_SOURCES})
target_include_directories(aetherwild_core PUBLIC include)
if(MSVC)
  target_compile_options(aetherwild_core PUBLIC /utf-8 /W4 /permissive-)
else()
  target_compile_options(aetherwild_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(aetherwild src/main.cpp)
target_link_libraries(aetherwild PRIVATE aetherwild_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(aetherwild_tests tests/core_tests.cpp)
  target_link_libraries(aetherwild_tests PRIVATE aetherwild_core)
  add_test(NAME core COMMAND aetherwild_tests)
endif()

add_executable(aetherwild_workstation apps/workstation.cpp)
target_link_libraries(aetherwild_workstation PRIVATE aetherwild_core)
if(WIN32)
  target_link_libraries(aetherwild_workstation PRIVATE ws2_32)
endif()

if(BUILD_TESTING)
 add_executable(aetherwild_runtime_tests tests/runtime_tests.cpp)
 target_link_libraries(aetherwild_runtime_tests PRIVATE aetherwild_core)
 add_test(NAME workstation COMMAND aetherwild_runtime_tests)
endif()


if(BUILD_TESTING)
 add_executable(aetherwild_content_tests tests/content_tests.cpp)
 target_link_libraries(aetherwild_content_tests PRIVATE aetherwild_core)
 add_test(NAME content COMMAND aetherwild_content_tests)
endif()
